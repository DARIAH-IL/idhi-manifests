IDHI LinkML schema

# yaml-language-server: $schema=https://linkml.io/linkml-model/linkml_model/jsonschema/meta.schema.json

# =====================================================================
# Israeli Digital Humanities Index (IDHI) — LinkML schema
# Documentation (modeling guide, identifier scheme, build workflow):
# see README.md at the repository root.
# =====================================================================
id: https://idhi_placeholder/linkml/idhi
name: idhi
title: Israeli Digital Humanities Index Schema
description: >-
  A LinkML schema for an index of Digital Humanities (DH) research in Israel. Entities and properties reuse established vocabularies (FOAF, schema.org, Dublin Core Terms, BIBO, DCAT, EDM, SKOS, CERIF); DH projects are classified with TaDiRAH 2.0 research-activity concepts (dynamic enum over the full taxonomy). Free text supports English, Hebrew and Arabic. All entities are identified by IDHI URNs (idhi:<class>:<shortid>); external PIDs are supplementary.
license: https://creativecommons.org/publicdomain/zero/1.0/
version: 0.1.0

# ------------------------- PREFIXES ----------------------------------
prefixes:
  idhi: https://idhi_placeholder/linkml/ # Israeli Digital Humanities Index (this schema's own namespace)
  linkml: https://w3id.org/linkml/ # Linked Data Modeling Language
  rdf: http://www.w3.org/1999/02/22-rdf-syntax-ns# # Resource Description Framework
  dcterms: http://purl.org/dc/terms/ # Dublin Core Metadata Terms
  foaf: https://xmlns.com/foaf/spec/#term_ # Friend of a Friend
  bibo: http://purl.org/ontology/bibo/ # Bibliographic Ontology
  schema: http://schema.org/ # Schema.org
  skos: http://www.w3.org/2004/02/skos/core# # Simple Knowledge Organization System
  dcat: http://www.w3.org/ns/dcat# # Data Catalog Vocabulary
  cerif: https://w3id.org/cerif/model# # Common European Research Information Format
  frapo: http://purl.org/cerif/frapo/ # Funding, Research Administration and Projects Ontology
  tadirah: https://vocabs.dariah.eu/tadirah/ # Taxonomy of Digital Research Activities in the Humanities
  coar: http://purl.org/coar/resource_type/ # Confederation of Open Access Repositories
  spdx: https://spdx.org/licenses/ # Software Package Data Exchange

default_prefix: idhi
default_range: string
default_curi_maps:
  - semweb_context

imports:
  - linkml:types

# ------------------------- SETTINGS ----------------------------------
# Reusable regex fragments for structured_pattern validation.
settings:
  shortid: "[0-9a-z]{4,12}"
  orcid: "\\d{4}-\\d{4}-\\d{4}-\\d{3}[0-9X]"
  ror: "0[a-hj-km-np-tv-z0-9]{6}[0-9]{2}"
  doi: ".+"

# ------------------------- SUBSETS ----------------------------------
subsets:
  toplevel_entity:
    description: >-
      Primary first-class entities in the IDHI index — those that carry their own IDHI URN and are listed directly in IndexContainer.

# ========================= CLASSES ===================================
classes:
  # ---- Abstract superclass -----------------------------------------
  Entity:
    abstract: true
    description: >-
      Root class for any identifiable IDHI entity.
      Provides the IDHI URN id, multilingual description, homepage, same_as links and tags.
      Never instantiated directly; every concrete entity inherits from it and constrains the id's class token via slot_usage.
    class_uri: schema:Thing
    slots:
      - type
      - id
      - description
      - homepage
      - same_as
      - tags

  # ---- Multilingual text (rdf:langString convention) ---------------
  LangString:
    description: >-
      A single language-tagged text value.
      Instances are combined in a multivalued slot to give English/Hebrew/Arabic variants of one field.
      Use one LangString per language; do not repeat a language within the same field.
    class_uri: rdf:langString
    slots:
      - language
      - value

  # ---- Agents ------------------------------------------------------
  Agent:
    abstract: true
    is_a: Entity
    description: >-
      A person or an organization.
    class_uri: foaf:Agent

  Person:
    is_a: Agent
    description: >-
      A human agent in the DH index: researcher, developer, librarian, student, etc.
    class_uri: foaf:Person
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Person"
      id:
        structured_pattern:
          syntax: "^idhi:person:{shortid}$"
          interpolated: true
    slots:
      - given_name
      - family_name
      - orcid
      - emails
      - affiliations # reified Person-Organization links
      - project_participations # reified Person-Project links
      - authorships # reified Person-Publication links

  Organization:
    is_a: Agent
    description: >-
      An organization of any kind.
      Its kind (academic institution, GLAM, research center, funder, company, non-profit) is given by organization_type.
      All organizations use the idhi:organization:<shortid> URN form.
    class_uri: foaf:Organization
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Organization"
      id:
        structured_pattern:
          syntax: "^idhi:organization:{shortid}$"
          interpolated: true
    slots:
      - name
      - ror
      - organization_type
      - parent_organization
      - location
      - address
      - additional_urls
      - contact_email

  Facility:
    is_a: Entity
    description: >-
      A physical or virtual facility such as a DH lab, digitization studio or research infrastructure, affiliated with one or more organizations.
      Use Facility when the unit offers services/tools and has its own identity distinct from its host organization; otherwise just use the Organization.
    class_uri: cerif:Facility
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Facility"
      id:
        structured_pattern:
          syntax: "^idhi:facility:{shortid}$"
          interpolated: true
    slots:
      - name
      - facility_affiliations # reified Facility-Organization links
      - location
      - address
      - services_offered
      - tools_provided
      - additional_urls
      - contact_email

  Project:
    is_a: Entity
    description: >-
      A Digital Humanities research project, classified by its research activities and disciplines.
      This is the central entity of the index; people, organizations, outputs and studied periods/places all hang off it.
    class_uri: foaf:Project
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Project"
      id:
        structured_pattern:
          syntax: "^idhi:project:{shortid}$"
          interpolated: true
    slots:
      - name
      - digital_humanities_activities
      - research_disciplines
      - start_date
      - end_date
      - studied_periods # historical period(s) studied
      - studied_places
      - project_participations
      - organization_roles # reified Organization-Project links
      - outputs_tools
      - outputs_datasets
      - outputs_publications
      - funding
      - additional_urls
      - contact_email

  Tool:
    is_a: Entity
    description: >-
      A reusable software tool, typically produced by a project.
      Use Tool for software that others can install, run or call; for a human-mediated offering use Service instead.
    class_uri: schema:SoftwareApplication
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Tool"
      id:
        structured_pattern:
          syntax: "^idhi:tool:{shortid}$"
          interpolated: true
    slots:
      - name
      - tool_type
      - code_repository
      - programming_language
      - license
      - documentation_url
      - additional_urls
      - contact_email
      - digital_humanities_activities

  Service:
    is_a: Entity
    description: >-
      A reusable, human- or organization-mediated service offered by a facility or organization (e.g., digitization on demand, OCR consulting, data curation support).
      Use Service when the offering requires the provider to act; use Tool for self-service software.
    class_uri: schema:Service
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Service"
      id:
        structured_pattern:
          syntax: "^idhi:service:{shortid}$"
          interpolated: true
    slots:
      - name
      - service_type
      - provider
      - documentation_url
      - additional_urls
      - contact_email
      - digital_humanities_activities

  Publication:
    is_a: Entity
    description: >-
      An academic publication: journal article, book, chapter, conference paper, thesis, report, etc.
      The precise kind is given by publication_type.
    class_uri: bibo:Document
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Publication"
      id:
        structured_pattern:
          syntax: "^idhi:publication:{shortid}$"
          interpolated: true
    slots:
      - name
      - doi
      - publication_type
      - authorships
      - date_issued
      - published_in
      - publisher
      - part_of
      - presented_at

  Event:
    is_a: Entity
    description: >-
      A scholarly event: conference, workshop, seminar, lecture, hackathon or exhibition.
      Use Event for time-bounded gatherings; recurring series should be modeled as one Event per occurrence.
    class_uri: schema:Event
    in_subset: [ toplevel_entity ]
    slot_usage:
      type:
        equals_string: "idhi:Event"
      id:
        structured_pattern:
          syntax: "^idhi:event:{shortid}$"
          interpolated: true
    slots:
      - name
      - event_type
      - start_date
      - end_date
      - location
      - address
      - additional_urls
      - contact_email

  Dataset:
    is_a: Entity
    description: >-
      A dataset, digital archive or catalog produced or curated by a project: corpora, databases, image collections, annotation sets and collections of metadata records.
      Use Dataset for both research data and catalogs that describe other resources; a catalog can link the datasets it aggregates through datasets.
    class_uri: dcat:Dataset
    slot_usage:
      id:
        structured_pattern:
          syntax: "^idhi:dataset:{shortid}$"
          interpolated: true
    slots:
      - name
      - datasets
      - publisher
      - license
      - date_issued
      - distribution_url
      - themes

  # ---- Reified relationship (role) classes -------------------------
  # Relationship objects are inlined and carry no IDHI URN of their own.
  Relationship:
    abstract: true
    description: >-
      Abstract base for reified, role-bearing relationships.
      A Relationship is used instead of a direct edge whenever the connection between two entities needs its own metadata: a role and/or a validity interval.
      Never instantiated directly.
    class_uri: schema:Role
    slots:
      - start_date
      - end_date

  ProjectParticipation:
    is_a: Relationship
    description: >-
      A person's participation in a project. Create one instance per (person, project, role) combination; if a person changed roles over time, create one instance per role with start/end dates.
    class_uri: cerif:Project_Person
    slots:
      - participant # -> Person
      - project # -> Project
      - participation_role

  Affiliation:
    is_a: Relationship
    description: >-
      A person's employment or membership at an organization, with a position and dates.
      Use for the person's institutional home(s), independent of any project.
    class_uri: cerif:Person_OrganisationUnit
    slots:
      - member # -> Person
      - organization # -> Organization
      - affiliation_role

  OrganizationProjectRole:
    is_a: Relationship
    description: >-
      An organization's engagement in a project.
      Use one instance per role: an organization that both hosts and funds a project gets two instances.
    class_uri: cerif:Project_OrganisationUnit
    slots:
      - organization # -> Organization
      - project # -> Project
      - org_project_role

  Funding:
    description: >-
      A funding award for a project, identifying the organization that provides it and its amount when public.
      It is inlined within the funded Project and has no independent ID.
    class_uri: schema:MonetaryGrant
    slots:
      - funding_organization
      - funding_amount

  Authorship:
    is_a: Relationship
    description: >-
      A person's contribution to a publication, with author order and role.
      Create one instance per (person, publication); author_order preserves the byline sequence (1 = first author).
    class_uri: schema:Role
    slots:
      - author # -> Person
      - publication # -> Publication
      - author_order
      - authorship_role

  FacilityAffiliation:
    is_a: Relationship
    description: >-
      A facility's affiliation with an organization.
      Use one instance per hosting/owning organization; joint labs get several.
    class_uri: cerif:Facility_OrganisationUnit
    slots:
      - facility # -> Facility
      - organization # -> Organization

  # ---- Container / tree root ---------------------------------------
  IndexContainer:
    tree_root: true
    description: >-
      Top-level holder for all IDHI records.
      Big entities live exactly once in these lists; everything else references them by their IDHI URN.
    attributes:
      persons:
        description: All Person records in the index.
        multivalued: true
        inlined_as_list: true
        range: Person
      organizations:
        description: All Organization records in the index.
        multivalued: true
        inlined_as_list: true
        range: Organization
      projects:
        description: All Project records in the index.
        multivalued: true
        inlined_as_list: true
        range: Project
      facilities:
        description: All Facility records in the index.
        multivalued: true
        inlined_as_list: true
        range: Facility
      tools:
        description: All Tool records in the index.
        multivalued: true
        inlined_as_list: true
        range: Tool
      services:
        description: All Service records in the index.
        multivalued: true
        inlined_as_list: true
        range: Service
      publications:
        description: All Publication records in the index.
        multivalued: true
        inlined_as_list: true
        range: Publication
      events:
        description: All Event records in the index.
        multivalued: true
        inlined_as_list: true
        range: Event
      datasets:
        description: All Dataset records in the index, including digital archives and catalogs.
        multivalued: true
        inlined_as_list: true
        range: Dataset

# ========================= SLOTS =====================================
slots:
  # --- identity / shared ---
  type:
    required: true
    range: curie
    description: Discriminator identifying the record's class; used for polymorphic serialization and deserialization.
    slot_uri: rdf:type
  id:
    identifier: true
    range: string
    description: >-
      The entity's primary identifier: an IDHI URN of the form
        idhi:<class name>:<random short alphanumeric id>
      e.g. idhi:person:x7k2m9 or idhi:project:a83bq1.
      Minted by IDHI at record creation and never reused or changed.
      The class token is the lowercase snake_case class name; each concrete class enforces its own token via slot_usage.
      External identifiers (ORCID, ROR, DOI...) are supplementary and go in their dedicated slots — never here.
    slot_uri: dcterms:identifier
    pattern: "^idhi:[a-z_]+:[0-9a-z]{4,12}$"
  name:
    range: LangString
    multivalued: true
    minimum_cardinality: 1
    inlined_as_list: true
    description: >-
      Multilingual name/title. Provide at least one language; English, Hebrew and Arabic variants are each a separate LangString.
      Preferably a sortable name for organizations; for projects, tools and services, use the name the team itself uses.
    slot_uri: skos:prefLabel
  description:
    range: LangString
    multivalued: true
    inlined_as_list: true
    description: >-
      Multilingual free-text description (a few sentences aimed at index visitors, not internal notes).
    slot_uri: dcterms:description
  homepage:
    range: uri
    description: Public landing page of the entity, if one exists.
    slot_uri: foaf:homepage
  same_as:
    range: uri
    multivalued: true
    description: >-
      URIs of records in OTHER systems describing the same real-world entity (Wikidata, PeriodO, GeoNames...).
      Use for linked-data alignment, not for the entity's own pages (use homepage).
    slot_uri: schema:sameAs
  tags:
    range: string
    multivalued: true
    description: >-
      Free-text tags for discovery, filtering and grouping; usable on any top-level entity.
      Deliberately NOT a controlled enum, but prefer wording that matches a concept in an established ontology or thesaurus (e.g. Wikidata, Getty AAT, TaDiRAH) so tags can later be reconciled against it.
    slot_uri: dcat:keyword
  additional_urls:
    range: uri
    multivalued: true
    description: >-
      Further relevant web pages beyond the homepage (blog, social-media profile, registry entry, press coverage...).
      For records describing the same entity in other systems use same_as instead.
    slot_uri: foaf:page
  contact_email:
    description: >-
      A published contact address for the entity (office, team or service-desk mailbox). For a person's own addresses use 'emails'.
    slot_uri: schema:email

  # --- LangString ---
  language:
    range: LanguageEnum
    required: true
    description: BCP-47 language tag of the value (en, he or ar).
    slot_uri: dcterms:language
  value:
    range: string
    required: true
    description: A localized text, in the language given by 'language'.
    slot_uri: rdf:value

  # --- Person ---
  given_name:
    required: false
    description: Given (first) name, in the person's preferred romanization. Use with family_name when the person's name is conventionally expressed in that form.
    slot_uri: foaf:givenName
  family_name:
    required: false
    description: Family (last) name, in the person's preferred romanization. Use with given_name when the person's name is conventionally expressed in that form.
    slot_uri: foaf:familyName
  orcid:
    range: uri
    slot_uri: schema:identifier
    description: >-
      The person's persistent researcher identifier.
      It supplements the IDHI record id.
      Strongly recommended for every researcher; enables deduplication and linking to the scholarly record.
    structured_pattern:
      syntax: "https://orcid.org/{orcid}"
      interpolated: true
  emails:
    multivalued: true
    description: >-
      Contact email addresses (zero or more). Only record addresses the person has agreed to publish in the index.
    slot_uri: foaf:mbox
  affiliations:
    range: Affiliation
    multivalued: true
    inlined_as_list: true
    description: >-
      The person's institutional affiliations, as reified Affiliation objects (organization + position + dates).
      Use for employment or formal membership, NOT for project involvement — that goes in project_participations.
  project_participations:
    range: ProjectParticipation
    multivalued: true
    inlined_as_list: true
    description: >-
      The person's project involvements, as reified ProjectParticipation objects carrying the role (PI, developer...) and dates.
  authorships:
    range: Authorship
    multivalued: true
    inlined_as_list: true
    description: >-
      The person's publication contributions, as reified Authorship objects carrying byline order and role.

  # --- Organization ---
  ror:
    range: uri
    slot_uri: schema:identifier
    description: >-
      The organization's persistent registry identifier.
      It supplements the IDHI record id.
      Record it whenever the organization is registered in ROR — most universities and research institutes are.
    structured_pattern:
      syntax: "https://ror.org/{ror}"
      interpolated: true
  organization_type:
    range: OrganizationTypeEnum
    description: >-
      The kind of organization. Always set it.
    slot_uri: dcterms:type
  parent_organization:
    range: Organization
    description: >-
      The larger organization this one is part of (e.g. a department's university). Use for formal containment only; looser partnerships belong in relationship classes.
    slot_uri: schema:parentOrganization
  location:
    range: LangString
    multivalued: true
    inlined_as_list: true
    description: >-
      Place name where the organization, facility or event is physically situated (e.g. a city), as free multilingual text.
    slot_uri: schema:location

  # --- Facility ---
  facility_affiliations:
    range: FacilityAffiliation
    multivalued: true
    inlined_as_list: true
    description: >-
      The organization(s) hosting or owning this facility, as reified FacilityAffiliation objects with dates.
  services_offered:
    range: Service
    multivalued: true
    description: >-
      Services this facility offers to researchers.
      Reference Service records by id; the Service's own 'provider' may still point at the parent Organization.
  tools_provided:
    range: Tool
    multivalued: true
    description: >-
      Tools this facility maintains or gives access to (by id).
      Use for hosted instances and lab-maintained software, not for every tool staff members happen to use.

  # --- Project ---
  digital_humanities_activities:
    range: DigitalHumanitiesActivityEnum
    multivalued: true
    description: >-
      Digital-humanities research activities practiced in this project, tool or service.
      Prefer the most specific applicable activity; multiple values are expected.
      This is the primary DH-facet for discovery.
    slot_uri: dcterms:subject
  research_disciplines:
    range: LangString
    multivalued: true
    inlined_as_list: true
    description: >-
      Humanities discipline(s) of the project (history, linguistics, archaeology...).
      Free multilingual text for now; a controlled vocabulary is a planned upgrade.
    slot_uri: dcterms:subject
  studied_periods:
    range: LangString
    multivalued: true
    inlined_as_list: true
    description: >-
      Historical period(s) the project studies (e.g. Ottoman period), as free multilingual labels — distinct from the project's own start_date/end_date.
    slot_uri: dcterms:temporal
  studied_places:
    range: LangString
    multivalued: true
    inlined_as_list: true
    description: >-
      Geographic focus of the research (places studied), as free multilingual labels — distinct from where the project team sits.
    slot_uri: dcterms:spatial
  organization_roles:
    range: OrganizationProjectRole
    multivalued: true
    inlined_as_list: true
    description: >-
      Organizations engaged in the project, as reified OrganizationProjectRole objects (coordinator, partner, funder, host).
  outputs_tools:
    range: Tool
    multivalued: true
    description: Tools produced by this project (by id).
    slot_uri: schema:producer
  outputs_datasets:
    range: Dataset
    multivalued: true
    description: Datasets produced or curated by this project (by id).
  outputs_publications:
    range: Publication
    multivalued: true
    description: Publications resulting from this project (by id).
  funding:
    range: Funding
    multivalued: true
    inlined_as_list: true
    description: Funding awards received by the project. Use one entry for each funding organization and award.
  funding_organization:
    range: Organization
    required: true
    description: The organization that provides this funding award (by IDHI URN).
    slot_uri: schema:funder
